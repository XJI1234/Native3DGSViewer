cmake_minimum_required(VERSION 3.22.1)
project(Native3DGSConsumer C)
if(NOT DEFINED NATIVE3DGS_SDK_ROOT)
    message(FATAL_ERROR "Set NATIVE3DGS_SDK_ROOT to the extracted SDK directory")
endif()
set(Native3DGSAndroid_DIR "${NATIVE3DGS_SDK_ROOT}/cmake")
find_package(Native3DGSAndroid CONFIG REQUIRED)
add_executable(native3dgs_consumer main.c)
target_link_libraries(native3dgs_consumer PRIVATE Native3DGSAndroid::Engine)
