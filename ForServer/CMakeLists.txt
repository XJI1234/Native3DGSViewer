cmake_minimum_required(VERSION 3.22)
project(Native3DGSServer LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
option(GS_SERVER_CUDA "Build Linux NVIDIA renderer" ON)
option(GS_SERVER_SANITIZERS "CPU Address/Undefined sanitizers" OFF)
set(GS_SERVER_CUDA_ARCHITECTURES "86-real;86-virtual" CACHE STRING "NVIDIA CUDA targets; set per deployment GPU")
if(WIN32)
    set(GS_SERVER_CUDA OFF)
endif()
set(ZSTD_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/third_party/zstd/build/cmake" "${CMAKE_BINARY_DIR}/zstd" EXCLUDE_FROM_ALL)
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/third_party/googletest" "${CMAKE_BINARY_DIR}/googletest" EXCLUDE_FROM_ALL)
add_library(gs_frame src/frame.cpp src/reference.cpp src/request.cpp)
target_include_directories(gs_frame PUBLIC include)
target_link_libraries(gs_frame PUBLIC libzstd_static)
if(WIN32)
    target_sources(gs_frame PRIVATE src/jpeg_windows.cpp)
    target_link_libraries(gs_frame PRIVATE windowscodecs ole32)
else()
    find_package(JPEG REQUIRED)
    target_sources(gs_frame PRIVATE src/jpeg_linux.cpp)
    target_link_libraries(gs_frame PRIVATE JPEG::JPEG)
endif()
if(MSVC)
    target_compile_options(gs_frame PRIVATE /W4 /permissive-)
else()
    target_compile_options(gs_frame PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(GS_SERVER_SANITIZERS AND NOT WIN32)
    target_compile_options(gs_frame PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(gs_frame PUBLIC -fsanitize=address,undefined)
endif()
enable_testing()
add_executable(frame_tests tests/frame_tests.cpp tests/image_v2_tests.cpp tests/request_tests.cpp)
target_link_libraries(frame_tests PRIVATE gs_frame GTest::gtest_main)
add_test(NAME FrameContracts COMMAND frame_tests)
add_executable(gs-frame src/frame_tool.cpp)
target_link_libraries(gs-frame PRIVATE gs_frame)
if(WIN32)
    add_library(gs_thin src/windows_renderer.cpp src/windows_http.cpp)
    target_link_libraries(gs_thin PUBLIC gs_frame PRIVATE d3d12 dxgi d3dcompiler winhttp user32)
    target_compile_definitions(gs_thin PRIVATE UNICODE _UNICODE)
    add_executable(gs-client src/windows_main.cpp)
    target_link_libraries(gs-client PRIVATE gs_thin psapi)
    add_executable(windows_tests tests/windows_tests.cpp tests/transport_tests.cpp)
    target_link_libraries(windows_tests PRIVATE gs_thin GTest::gtest_main ws2_32)
    add_test(NAME WindowsThinRenderer COMMAND windows_tests)
    set_tests_properties(WindowsThinRenderer PROPERTIES TIMEOUT 30)
endif()
if(GS_SERVER_CUDA)
    enable_language(CUDA)
    find_package(CUDAToolkit 12.6 REQUIRED)
    find_package(ZLIB REQUIRED)
    find_package(Boost 1.74 REQUIRED COMPONENTS system)
    add_library(gs_server_model
        "${ROOT}/third_party/spz/src/cc/load-spz.cc"
        "${ROOT}/third_party/spz/src/cc/splat-c-types.cc"
        "${ROOT}/third_party/spz/src/cc/splat-types.cc"
        "${ROOT}/src/model-io/common/probe.cpp"
        "${ROOT}/src/model-io/normalize/normalize.cpp"
        "${ROOT}/src/model-io/codecs/decode.cpp"
        src/model.cpp)
    target_include_directories(gs_server_model PUBLIC include "${ROOT}/include" PRIVATE "${ROOT}/third_party/spz/src/cc")
    target_link_libraries(gs_server_model PUBLIC libzstd_static ZLIB::ZLIB)
    target_compile_options(gs_server_model PRIVATE -include cmath)
    add_library(gs_cuda src/cuda_renderer.cu)
    target_include_directories(gs_cuda PUBLIC include "${ROOT}/include")
    set_target_properties(gs_cuda PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON CUDA_ARCHITECTURES "${GS_SERVER_CUDA_ARCHITECTURES}")
    target_compile_options(gs_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
    target_link_libraries(gs_cuda PUBLIC gs_frame gs_server_model CUDA::cudart)
    add_executable(cuda_tests tests/cuda_tests.cpp)
    target_link_libraries(cuda_tests PRIVATE gs_cuda GTest::gtest_main)
    add_test(NAME CudaCorrectness COMMAND cuda_tests)
    add_executable(gs-server src/server_main.cpp)
    target_link_libraries(gs-server PRIVATE gs_cuda Boost::system)
endif()
